## Review Focus

1. A truncated or blank last line in a history file (writer mid-append) — reading must skip it, not crash. Test in Task 2.
2. Old activity rows missing `tool`/`args`/`ok`, or with `args` as a JSON string instead of a dict — report and classifier must tolerate. Tests in Tasks 2 and 3.
3. Tool callables wrapped with `functools.wraps` or taking `**kwargs` — round validation must accept. Test in Task 4.
4. Missing rounds library dir or corrupt round JSON — `warm_start` returns the valid remainder, never raises. Test in Task 4.
5. Compound shell commands (`cd x && git diff`, `env A=1 rg foo`, `pytest ... | tail`) — classifier looks through `cd`/`env` prefixes and `&&`/`;` chains but does not flag pipelines whose first segment is legit. Test in Task 2.

```python
import re
def _loads(s):
    try: return json.loads(s)
    except (json.JSONDecodeError, TypeError): return None

def read_history(path=RAMABANA_HISTORY, session=None):
    "Read completed Ramabana turns in append order, skipping partial lines."
    r = L(Path(path).expanduser().readlines()).map(_loads).filter(lambda x: isinstance(x, dict))
    return r if not session else r.filter(lambda x: x.get('session')==session)

def tool_args(action):
    "The call arguments of `action` as a dict."
    a = action.get('args')
    if isinstance(a, str): a = _loads(a)
    return a if isinstance(a, dict) else {}

SHELL_TOOLS = (
    (r'git\s+status\b', 'git_status'), (r'git\s+diff\b', 'git_diff'), (r'git\s+(log|show)\b', 'git_log'),
    (r'git\s+commit\b', 'git_commit'), (r'git\s+(checkout|switch)\b', 'git_checkout'), (r'git\s+stash\b', 'git_stash'),
    (r'git\s+(push|pull|fetch)\b', 'git_remote'), (r'(rg|grep|ag)\b', 'grep'), (r'(find|fd)\b', 'ls'), (r'(ls|tree)\b', 'ls'),
    (r'(sed|perl)\s+-p?i\b', 'replace_text'), (r'(cat\s*>|tee\b)', 'create_file'), (r'sed\s+-n\b|(cat|head|tail|nl)\b', 'view_file'),
    (r'(curl|wget)\b', 'read_url'))
_PREFIX = re.compile(r'^(env(\s+-u\s+\S+|\s+\w+=\S+)*|\w+=\S+)\s+')

def _segments(command):
    for s in re.split(r'&&|;|\|\|', command):
        s = _PREFIX.sub('', s.strip())
        if s and not s.startswith('cd '): yield s.split('|')[0].strip()

def bypass_tool(command, offered=None):
    "The dedicated tool `command` stands in for, if any."
    if re.search(r'&\s*$', command.strip()) and not command.strip().endswith('&&'): hit = 'run_shell_bg'
    else: hit = next((t for s in _segments(command) for rx,t in SHELL_TOOLS if re.match(rx, s)), None)
    return hit if hit and (offered is None or hit in offered) else None
```

```python
import inspect
ROUNDS_DIR = xdg_config_home()/'drona/rounds'
SEEDS_DIR = Path(__file__).parent/'seeds'

def call_valid(args, fn):
    "Would `fn` accept `args` as keyword arguments?"
    try: inspect.signature(fn).bind(**args); return True
    except (TypeError, ValueError): return False

def as_tools(tools): return dict(tools) if isinstance(tools, dict) else {f.__name__: f for f in tools}
def round_calls(rnd): return [(p['name'], p.get('arguments') or {}) for m in rnd['history'] for p in m['content'] if p['type']=='tool_use']
def round_valid(rnd, tools): return all(n in tools and call_valid(a, tools[n]) for n,a in round_calls(rnd))

def load_rounds(dirs=None):
    "Accepted rounds from the library and packaged seeds."
    res = []
    for d in dirs or [ROUNDS_DIR, SEEDS_DIR]:
        for p in sorted(Path(d).glob('*.json')) if Path(d).exists() else []:
            r = _loads(p.read_text())
            if isinstance(r, dict) and r.get('meta', {}).get('status') == 'accepted': res.append(r)
    return res

def warm_start(tools=None, model=None, limit=2, dirs=None):
    "Canonical Urai history from accepted rounds that fit the offered tools."
    rs = load_rounds(dirs)
    if tools is not None: rs = [r for r in rs if round_valid(r, as_tools(tools))]
    rs = sorted(rs, key=lambda r: r['meta'].get('model') != model)[:limit]
    return [m for r in rs for m in round_msgs(r)]
```
