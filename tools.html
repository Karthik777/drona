tools

Count which tools agents call, how often the calls fail, and when the shell stands in for a dedicated tool.

dhrona-tools measures tool use before and after you change an agent’s tool set, for example when you remove tools nobody uses. It reads the Ramabana and Leela history files and reports, for each tool, the calls, failures and refusals. It then lists the shell commands that did a dedicated tool’s job, and gives the same counts for each model.

Report

Each turn’s at is a time in seconds since the epoch, so since accepts either that or an ISO date. Older rows may lack tool, ok or args, or hold args as JSON text, and the report accepts all of them. Failures are grouped by the first line of their detail with every number replaced by N, so the same error at different line numbers counts as one. A refused approval, which is an ask row with ok false or a detail starting Denied by human operator, counts as a denial with its reason, not as a failure. fmt_report prints the report as a table.

tool_report

def tool_report(
    turns, # history turns from `read_history`
    since:NoneType=None, # epoch or ISO date lower bound on `at`
    model:NoneType=None, # restrict to one model name
):

Count calls, failures and denials per tool, shell bypasses, and the same split per model.

AT_OLD, AT_NEW, SINCE_BETWEEN = 1786431016.4, 1790392050.7, 1788000000.0
turns = [
  {'model': 'm1', 'at': AT_OLD, 'activity': [{'tool': 'run_shell', 'args': {'command': 'git diff'}, 'ok': True}]},
  {'model': 'm2', 'at': AT_NEW, 'activity': [
     {'tool': 'edit_cell', 'args': '{}', 'ok': False, 'detail': 'JSONDecodeError: Extra data at line 3'},
     {'tool': 'edit_cell', 'args': {}, 'ok': False, 'detail': 'JSONDecodeError: Extra data at line 9'},
     {'tool': 'view_file', 'ok': True}, {'ok': True}]}]
r = tool_report(turns)
assert r['calls'] == 5 and r['tools']['edit_cell']['fails'] == 2 and r['tools']['edit_cell']['fail_rate'] == 1.0
assert r['tools']['edit_cell']['failures'][0][1] == 2
assert r['tools']['edit_cell']['failures'][0][0] == 'JSONDecodeError: Extra data at line N'
assert list(r['tools']) == ['edit_cell', 'run_shell', 'view_file', '?']
assert r['tools']['run_shell']['fails'] == 0 and r['tools']['run_shell']['failures'] == []
assert r['bypass'] == {'git_diff': 1} and r['models']['m1'] == {'calls': 1, 'shell': 1, 'bypass': 1}
assert r['models']['m2'] == {'calls': 4, 'shell': 0, 'bypass': 0}
assert tool_report(turns, since=SINCE_BETWEEN)['calls'] == 4
assert tool_report(turns, since='2026-09-01')['calls'] == 4 and tool_report(turns, since=str(SINCE_BETWEEN))['calls'] == 4
assert tool_report(turns, model='m1')['calls'] == 1
assert tool_report([])['calls'] == 0 and fmt_report(tool_report([]))
assert 'edit_cell' in fmt_report(r) and 'git_diff=1' in fmt_report(r) and 'm1:' in fmt_report(r)
assert dumps(r)
assert tool_report([{'activity': [{'tool': 'x', 'detail': 'boom'}]}])['tools']['x']['fails'] == 0
assert r['denials'] == {} and r['denial_reasons'] == [] and 'denials: none' in fmt_report(r)
refused = [{'tool': 'replace_text', 'args': {'path': 'x.py'}, 'ok': False, 'kind': 'ask', 'detail': 'Denied by human operator. Reason given: no answer after 300s'},
           {'tool': 'replace_text', 'args': {'path': 'x.py'}, 'ok': False, 'detail': 'Denied by human operator. Reason given: no answer after 900s'},
           {'tool': 'run_shell', 'args': {'command': 'rm -rf x'}, 'ok': False, 'kind': 'ask', 'detail': 'Denied by human operator'}]
rd = tool_report(turns + [{'model': 'm2', 'at': AT_NEW, 'activity': refused}])
assert rd['denials'] == {'replace_text': 2, 'run_shell': 1}
assert rd['denial_reasons'] == [['no answer after Ns', 2], ['no reason given', 1]]
assert rd['tools']['replace_text'] == {'calls': 2, 'fails': 0, 'fail_rate': 0.0, 'failures': []}
assert rd['calls'] == 8 and rd['models']['m2']['calls'] == 7
assert 'denials: replace_text=2, run_shell=1; no answer after Ns=2, no reason given=1' in fmt_report(rd)

Command line

dhrona-tools reads every known history by default. --history a,b picks the files, --since sets the start of the window, --model keeps one model, and --json prints the machine-readable form to save as a baseline.

A missing history path is named on stderr and skipped. When no file is left, or when --since is neither an epoch nor an ISO date, the command exits with one line instead of a traceback.

import io, contextlib, os
from unittest.mock import patch
from fastcore.test import expect_fail
with expect_fail(ValueError, '--since must be'): _epoch('yesterday')
with TemporaryDirectory() as d:
    p, missing, err = Path(d)/'h.jsonl', f'{d}/missing.jsonl', io.StringIO()
    p.write_text(_d(turns[0]) + '\n')

    assert 'missing.jsonl' in err.getvalue()
